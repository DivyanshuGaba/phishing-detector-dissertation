FILE train_distilbert.py
import pandas as pd
import torch
from torch.utils.data import Dataset, DataLoader
from transformers import DistilBertTokenizer, DistilBertForSequenceClassification
from torch.optim import AdamW
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
import time

#  Device setup 
if torch.backends.mps.is_available():
    device = torch.device("mps")
    print("Using Apple Silicon MPS")
elif torch.cuda.is_available():
    device = torch.device("cuda")
    print("Using NVIDIA CUDA")
else:
    device = torch.device("cpu")
    print("Using CPU")

#  Load data 
print("\nLoading data...")
train = pd.read_csv("train_combined.csv")
val = pd.read_csv("val_combined.csv")

train["text"] = train["subject"].fillna("") + " " + train["body"].fillna("")
val["text"] = val["subject"].fillna("") + " " + val["body"].fillna("")

#  Label encoding 
label2id = {"legitimate": 0, "phishing": 1}
train["label_id"] = train["label"].map(label2id)
val["label_id"] = val["label"].map(label2id)

print(f"Training samples: {len(train)}")
print(f"Validation samples: {len(val)}")

#  Tokeniser 
print("\nLoading DistilBERT tokeniser...")
tokeniser = DistilBertTokenizer.from_pretrained("distilbert-base-uncased")

#  Dataset class 
class EmailDataset(Dataset):
    def __init__(self, texts, labels, tokeniser, max_length=128):
        self.texts = texts
        self.labels = labels
        self.tokeniser = tokeniser
        self.max_length = max_length

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        encoding = self.tokeniser(
            str(self.texts[idx]),
            truncation=True,
            padding="max_length",
            max_length=self.max_length,
            return_tensors="pt"
        )
        return {
            "input_ids": encoding["input_ids"].squeeze(),
            "attention_mask": encoding["attention_mask"].squeeze(),
            "label": torch.tensor(self.labels[idx], dtype=torch.long)
        }

#  Create datasets 
train_dataset = EmailDataset(
    train["text"].tolist(),
    train["label_id"].tolist(),
    tokeniser
)
val_dataset = EmailDataset(
    val["text"].tolist(),
    val["label_id"].tolist(),
    tokeniser
)

#  Data loaders 
train_loader = DataLoader(train_dataset, batch_size=8, shuffle=True)
val_loader = DataLoader(val_dataset, batch_size=8, shuffle=False)

#  Load model 
print("\nLoading DistilBERT model...")
model = DistilBertForSequenceClassification.from_pretrained(
    "distilbert-base-uncased",
    num_labels=2
)
model = model.to(device)

#  Optimiser 
optimiser = AdamW(model.parameters(), lr=2e-5, weight_decay=0.01)

#  Training loop 
print("\nStarting training...")
EPOCHS = 3

for epoch in range(EPOCHS):
    model.train()
    total_loss = 0
    start_time = time.time()

    for batch_idx, batch in enumerate(train_loader):
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)
        labels = batch["label"].to(device)

        optimiser.zero_grad()
        outputs = model(
            input_ids=input_ids,
            attention_mask=attention_mask,
            labels=labels
        )
        loss = outputs.loss
        loss.backward()
        optimiser.step()
        total_loss += loss.item()

        if (batch_idx + 1) % 100 == 0:
            print(f"  Epoch {epoch+1} | Batch {batch_idx+1}/{len(train_loader)} | Loss: {loss.item():.4f}")

    avg_loss = total_loss / len(train_loader)
    elapsed = time.time() - start_time
    print(f"\nEpoch {epoch+1} complete | Avg Loss: {avg_loss:.4f} | Time: {elapsed:.1f}s")

    #  Validation 
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for batch in val_loader:
            input_ids = batch["input_ids"].to(device)
            attention_mask = batch["attention_mask"].to(device)
            labels = batch["label"].to(device)

            outputs = model(
                input_ids=input_ids,
                attention_mask=attention_mask
            )
            preds = torch.argmax(outputs.logits, dim=1)
            all_preds.extend(preds.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())

    accuracy  = accuracy_score(all_labels, all_preds)
    precision = precision_score(all_labels, all_preds)
    recall    = recall_score(all_labels, all_preds)
    f1        = f1_score(all_labels, all_preds)

    print(f"Validation | Accuracy: {accuracy:.4f} | Precision: {precision:.4f} | Recall: {recall:.4f} | F1: {f1:.4f}")

#  Save model 
print("\nSaving model...")
model.save_pretrained("distilbert_phishing")
tokeniser.save_pretrained("distilbert_phishing")
print("Model saved to distilbert_phishing/")
print("\nDone.")

def analyse(model_label, pred_col):
    section(f"{model_label} - FAILURE ANALYSIS (TEST SET)")

    df = test.copy()
    df["correct"] = df["label"] == df[pred_col]
    errors = df[~df["correct"]]

    total_errors = len(errors)
    fp = errors[(errors["label"] == "legitimate") & (errors[pred_col] == "phishing")]
    fn = errors[(errors["label"] == "phishing") & (errors[pred_col] == "legitimate")]

    print(f"\nTotal test emails : {len(df)}")
    print(f"Total misclassified: {total_errors}  ({fp.shape[0]} false positives, {fn.shape[0]} false negatives)")

    # FALSE POSITIVE RATE, SPLIT BY SOURCE
    print("\nFALSE POSITIVE RATE BY SOURCE  (legitimate emails wrongly flagged)")
    print(f"  {'Source':<24}{'Legitimate emails':>20}{'False positives':>17}{'FP rate':>9}")
    for source in ["synthetic", "real"]:
        legit_this_source = df[(df["label"] == "legitimate") & (df["source"] == source)]
        fp_this_source = fp[fp["source"] == source]
        n_legit = len(legit_this_source)
        n_fp = len(fp_this_source)
        rate = (n_fp / n_legit * 100) if n_legit else 0.0
        label = "Synthetic (modern)" if source == "synthetic" else "Real (Enron)"
        print(f"  {label:<24}{n_legit:>20}{n_fp:>17}{rate:>8.2f}%")

    # FALSE NEGATIVE RATE, SPLIT BY SOURCE
    print("\nFALSE NEGATIVE RATE BY SOURCE  (phishing emails missed)")
    print(f"  {'Source':<24}{'Phishing emails':>20}{'False negatives':>17}{'FN rate':>9}")
    for source in ["synthetic", "real"]:
        phish_this_source = df[(df["label"] == "phishing") & (df["source"] == source)]
        fn_this_source = fn[fn["source"] == source]
        n_phish = len(phish_this_source)
        n_fn = len(fn_this_source)
        rate = (n_fn / n_phish * 100) if n_phish else 0.0
        label = "Synthetic (modern)" if source == "synthetic" else "Real (historical)"
        print(f"  {label:<24}{n_phish:>20}{n_fn:>17}{rate:>8.2f}%")

    # ERROR BREAKDOWN BY CATEGORY
    print("\nERRORS BY CATEGORY")
    if len(errors):
        counts = errors["category"].value_counts()
        for cat, n in counts.items():
            print(f"  {cat:<30}{n:>10}")
    else:
        print("  No errors.")

    print()
    return {
        "total_errors": total_errors,
        "fp_synthetic_rate": (len(fp[fp["source"] == "synthetic"]) / max(len(df[(df["label"] == "legitimate") & (df["source"] == "synthetic")]), 1)) * 100,
        "fp_real_rate": (len(fp[fp["source"] == "real"]) / max(len(df[(df["label"] == "legitimate") & (df["source"] == "real")]), 1)) * 100,
        "fn_synthetic_rate": (len(fn[fn["source"] == "synthetic"]) / max(len(df[(df["label"] == "phishing") & (df["source"] == "synthetic")]), 1)) * 100,
        "fn_real_rate": (len(fn[fn["source"] == "real"]) / max(len(df[(df["label"] == "phishing") & (df["source"] == "real")]), 1)) * 100,
    }

def token_saliency(text):
    """Gradient x input attribution for one email.

    Returns the wordpiece tokens, their saliency scores, and the
    predicted label. Special and padding tokens are removed.
    """
    encoding = tokeniser(
        text, truncation=True, padding="max_length",
        max_length=128, return_tensors="pt"
    )
    input_ids = encoding["input_ids"].to(device)
    attention_mask = encoding["attention_mask"].to(device)

    embed_layer = db_model.get_input_embeddings()
    inputs_embeds = embed_layer(input_ids)
    inputs_embeds.retain_grad()

    outputs = db_model(inputs_embeds=inputs_embeds, attention_mask=attention_mask)
    pred_id = torch.argmax(outputs.logits, dim=1).item()

    db_model.zero_grad()
    outputs.logits[0, pred_id].backward()

    # gradient x input, summed over the embedding dimension
    grads = inputs_embeds.grad[0]
    scores = (grads * inputs_embeds[0]).sum(dim=-1).abs().detach().numpy()

    tokens = tokeniser.convert_ids_to_tokens(input_ids[0])
    mask = attention_mask[0].numpy()

    def is_word(token):
        return any(c.isalnum() for c in token)

    keep_tokens = []
    keep_scores = []
    for tok, score, m in zip(tokens, scores, mask):
        if m == 0 or tok in ("[CLS]", "[SEP]", "[PAD]") or not is_word(tok):
            continue
        keep_tokens.append(tok)
        keep_scores.append(float(score))

    return keep_tokens, np.array(keep_scores), id2label[pred_id]


def merge_wordpieces(tokens, scores):
    """Merge ## continuation pieces back into whole words.

    A word takes the maximum saliency of its constituent pieces, so a
    strong signal on any piece is not diluted by averaging.
    """
    words = []
    word_scores = []
    for tok, score in zip(tokens, scores):
        if tok.startswith("##") and words:
            words[-1] += tok[2:]
            word_scores[-1] = max(word_scores[-1], score)
        else:
            words.append(tok)
            word_scores.append(score)
    return words, np.array(word_scores)

def top_words(agg, n=TOP_N, min_count=2):
    """Mean saliency per word, requiring the word to appear more than once
    so that single high-scoring outliers do not dominate the ranking."""
    rows = [(w, float(np.mean(v)), len(v)) for w, v in agg.items() if len(v) >= min_count]
    rows.sort(key=lambda r: r[1], reverse=True)
    return rows[:n]

def plot_heatmap(entry, filename, per_row=8, max_words=48):
    """Render one email as a grid of tokens shaded by saliency."""
    words = entry["words"][:max_words]
    scores = entry["scores"][:max_words]

    rows = int(np.ceil(len(words) / per_row))
    grid = np.zeros((rows, per_row))
    labels = [["" for _ in range(per_row)] for _ in range(rows)]

    for i, (w, s) in enumerate(zip(words, scores)):
        r, c = divmod(i, per_row)
        grid[r, c] = s
        labels[r][c] = w

    fig, ax = plt.subplots(figsize=(per_row * 1.7, rows * 0.75))
    im = ax.imshow(grid, cmap="Reds", aspect="auto", vmin=0, vmax=1)

    for r in range(rows):
        for c in range(per_row):
            if labels[r][c]:
                shade = "white" if grid[r, c] > 0.6 else "black"
                ax.text(c, r, labels[r][c], ha="center", va="center",
                        fontsize=8, color=shade)

    ax.set_xticks([])
    ax.set_yticks([])
    title = (f"True: {entry['true_label']}  |  Predicted: {entry['predicted']}  "
             f"|  Category: {entry['category']}")
    ax.set_title(title, fontsize=10)
    fig.colorbar(im, ax=ax, label="Normalised saliency", fraction=0.02)
    plt.tight_layout()
    plt.savefig(filename, dpi=150)
    plt.close()

import argparse
import os
import sys
import time
import pickle
import numpy as np
import pandas as pd
import torch
from transformers import DistilBertTokenizer, DistilBertForSequenceClassification

DIRECTIONS_CACHE = "rf_feature_directions.csv"
WIDTH = 64


def section(text):
    print(text.center(WIDTH))


# INPUT HANDLING
def get_input_text(args):
    if args.file:
        with open(args.file, "r", encoding="utf-8", errors="ignore") as f:
            raw = f.read()
    elif args.text:
        raw = args.text
    else:
        print("Paste the email text below, then press Ctrl-D (Mac/Linux) "
              "or Ctrl-Z then Enter (Windows) to finish:\n")
        raw = sys.stdin.read()

    raw = raw.strip()
    subject = args.subject or ""
    body = raw

    # allow "Subject: " as the first line if no subject was given
    lines = raw.split("\n", 1)
    if not subject and lines[0].lower().startswith("subject:"):
        subject = lines[0][len("subject:"):].strip()
        body = lines[1].strip() if len(lines) > 1 else ""

    if not raw:
        print("No email text provided.")
        sys.exit(1)

    return subject, body


# RANDOM FOREST - LOAD + EXPLAIN
def load_rf():
    with open("rf_pipeline_combined.pkl", "rb") as f:
        pipeline = pickle.load(f)
    return pipeline["vectoriser"], pipeline["model"]


def build_or_load_directions(vectoriser, rf_model):
    """Direction (which class each feature points toward) isn't part of
    Gini importance itself, so it's derived once from the test set and
    cached to disk. Degrades gracefully if the cache and the test set
    are both unavailable - importance is still shown, just not direction.
    """
    if os.path.exists(DIRECTIONS_CACHE):
        return pd.read_csv(DIRECTIONS_CACHE).set_index("feature")

    if not os.path.exists("test_combined.csv"):
        return None

    print("Building feature-direction reference (first run only)...")
    test = pd.read_csv("test_combined.csv")
    test["text"] = test["subject"].fillna("") + " " + test["body"].fillna("")

    dense = vectoriser.transform(test["text"]).toarray()
    is_phishing = (test["label"] == "phishing").values
    names = vectoriser.get_feature_names_out()
    importances = rf_model.feature_importances_

    mean_phish = dense[is_phishing].mean(axis=0)
    mean_legit = dense[~is_phishing].mean(axis=0)
    direction = np.where(mean_phish > mean_legit, "phishing", "legitimate")

    df = pd.DataFrame({
        "feature": names,
        "importance": importances,
        "direction": direction
    }).set_index("feature")
    df.to_csv(DIRECTIONS_CACHE)
    print(f"Saved: {DIRECTIONS_CACHE}\n")
    return df


def explain_rf(vectoriser, rf_model, directions, text, top_n=5):
    """Of the features that actually appear in this email, return the
    top N by global Gini importance - a per-email explanation grounded
    in the model's overall learned importances."""
    vec = vectoriser.transform([text])
    present_idx = vec.nonzero()[1]
    if len(present_idx) == 0:
        return []

    names = vectoriser.get_feature_names_out()
    importances = rf_model.feature_importances_

    rows = []
    for idx in present_idx:
        name = names[idx]
        imp = importances[idx]
        if directions is not None and name in directions.index:
            direction = directions.loc[name, "direction"]
        else:
            direction = "unknown"
        rows.append((name, imp, direction))

    rows.sort(key=lambda r: r[1], reverse=True)
    return rows[:top_n]


def predict_rf(vectoriser, rf_model, text, n_runs=5):
    features = vectoriser.transform([text])

    rf_model.predict(features)
    timings = []
    for _ in range(n_runs):
        start = time.perf_counter()
        pred = rf_model.predict(features)[0]
        end = time.perf_counter()
        timings.append((end - start) * 1000)
    proba = rf_model.predict_proba(features)[0]
    classes = list(rf_model.classes_)
    conf = proba[classes.index(pred)]
    return pred, conf, np.mean(timings)


# DISTILBERT - LOAD + EXPLAIN
def load_distilbert():
    device = torch.device("cpu")  # single-email latency, consistent across machines
    tokeniser = DistilBertTokenizer.from_pretrained("distilbert_phishing")
    model = DistilBertForSequenceClassification.from_pretrained("distilbert_phishing")
    model.to(device)
    model.eval()
    return tokeniser, model, device


def predict_and_explain_db(tokeniser, model, device, text, top_n=5, n_runs=5):
    encoding = tokeniser(text, truncation=True, padding="max_length",
                          max_length=128, return_tensors="pt")
    input_ids = encoding["input_ids"].to(device)
    attention_mask = encoding["attention_mask"].to(device)

    # warm-up + averaged latency, same fix as the RF side
    with torch.no_grad():
        model(input_ids=input_ids, attention_mask=attention_mask)
    timings = []
    with torch.no_grad():
        for _ in range(n_runs):
            start = time.perf_counter()
            model(input_ids=input_ids, attention_mask=attention_mask)
            end = time.perf_counter()
            timings.append((end - start) * 1000)

    # separate pass with grad enabled for attribution
    embed_layer = model.get_input_embeddings()
    inputs_embeds = embed_layer(input_ids)
    inputs_embeds.retain_grad()

    outputs = model(inputs_embeds=inputs_embeds, attention_mask=attention_mask)
    probs = torch.softmax(outputs.logits, dim=1)[0]
    pred_id = torch.argmax(probs).item()
    conf = probs[pred_id].item()

    model.zero_grad()
    outputs.logits[0, pred_id].backward()
    grads = inputs_embeds.grad[0]
    scores = (grads * inputs_embeds[0]).sum(dim=-1).abs().detach().numpy()

    tokens = tokeniser.convert_ids_to_tokens(input_ids[0])
    mask = attention_mask[0].numpy()

    words, word_scores = [], []
    for tok, score, m in zip(tokens, scores, mask):
        if m == 0 or tok in ("[CLS]", "[SEP]", "[PAD]"):
            continue
        if tok.startswith("##") and words:
            words[-1] += tok[2:]
            word_scores[-1] = max(word_scores[-1], float(score))
        else:
            words.append(tok)
            word_scores.append(float(score))

    def is_word(token):
        return any(c.isalnum() for c in token)

    ranked = sorted(
        [(w, s) for w, s in zip(words, word_scores) if is_word(w)],
        key=lambda r: r[1], reverse=True
    )
    id2label = {0: "legitimate", 1: "phishing"}
    return id2label[pred_id], conf, np.mean(timings), ranked[:top_n]


# MAIN
def main():
    parser = argparse.ArgumentParser(
        description="Compare Random Forest and DistilBERT on a single email, "
                     "with a short explanation of each prediction."
    )
    parser.add_argument("--file", type=str, help="Path to a .txt file containing the email")
    parser.add_argument("--text", type=str, help="Raw email text")
    parser.add_argument("--subject", type=str, default="", help="Optional subject line")
    args = parser.parse_args()

    subject, body = get_input_text(args)
    text = (subject + " " + body).strip()

    section("PHISHING DETECTION - EMAIL ANALYSIS")
    print(f"\n  Subject : {subject if subject else '(none provided)'}")
    preview = body[:100] + ("..." if len(body) > 100 else "")
    print(f"  Body    : {preview}\n")

    print("\nLoading models...")
    vectoriser, rf_model = load_rf()
    directions = build_or_load_directions(vectoriser, rf_model)
    tokeniser, db_model, device = load_distilbert()

    # RANDOM FOREST
    rf_pred, rf_conf, rf_latency = predict_rf(vectoriser, rf_model, text)
    rf_top = explain_rf(vectoriser, rf_model, directions, text)

    print("\nRANDOM FOREST")
    print(f"  Prediction  : {rf_pred}")
    print(f"  Confidence  : {rf_conf:.4f}")
    print(f"  Latency     : {rf_latency:.3f} ms  (5 run average, warmup excluded)")
    if rf_top:
        print("\n  Top contributing features (present in this email):")
        for i, (feat, imp, direction) in enumerate(rf_top, start=1):
            print(f"    {i}. {feat:<20} importance {imp:.4f}  -> {direction}")
    else:
        print("\n  No recognised vocabulary from training found in this email.")

    # DISTILBERT
    db_pred, db_conf, db_latency, db_top = predict_and_explain_db(
        tokeniser, db_model, device, text
    )

    print("\nDISTILBERT")
    print(f"  Prediction  : {db_pred}")
    print(f"  Confidence  : {db_conf:.4f}")
    print(f"  Latency     : {db_latency:.3f} ms  (5 run average, warmup excluded)")
    if db_top:
        print("\n  Top salient tokens (gradient x input):")
        for i, (tok, score) in enumerate(db_top, start=1):
            print(f"    {i}. {tok:<20} saliency {score:.4f}")

    # SUMMARY
    print()
    section("SUMMARY")
    if rf_pred == db_pred:
        print(f"\n  Both models agree: {rf_pred.upper()}")
    else:
        print(f"\n  Models disagree:")
        print(f"    Random Forest -> {rf_pred}")
        print(f"    DistilBERT    -> {db_pred}")
    print(f"\n  Speed: Random Forest is {db_latency / rf_latency:.1f}x faster "
          f"than DistilBERT on this email")
    print()


if __name__ == "__main__":
    main()

if torch.backends.mps.is_available():
    device = torch.device("mps")
elif torch.cuda.is_available():
    device = torch.device("cuda")
else:
    device = torch.device("cpu")

tokenizer = DistilBertTokenizer.from_pretrained("distilbert_phishing")
db_model = DistilBertForSequenceClassification.from_pretrained("distilbert_phishing")
db_model.to(device)
db_model.eval()

db_preds = []
batch_size = 16
texts = val["text"].tolist()

with torch.no_grad():
    for i in range(0, len(texts), batch_size):
        batch = texts[i:i + batch_size]
        encodings = tokenizer(batch, truncation=True, padding="max_length", max_length=128, return_tensors="pt")
        encodings = {k: v.to(device) for k, v in encodings.items()}
        outputs = db_model(**encodings)
        batch_preds = torch.argmax(outputs.logits, dim=1).cpu().tolist()
        db_preds.extend(batch_preds)

db_accuracy = accuracy_score(val_binary, db_preds)
db_precision = precision_score(val_binary, db_preds)
db_recall = recall_score(val_binary, db_preds)
db_f1 = f1_score(val_binary, db_preds)

@st.cache_resource(show_spinner="Loading models (first run only)...")
def get_models():
    vectoriser, rf_model = load_rf()
    directions = build_or_load_directions(vectoriser, rf_model)
    tokeniser, db_model, device = load_distilbert()
    return vectoriser, rf_model, directions, tokeniser, db_model, device

if "run_body" in st.session_state:

    if not text:
        st.error("No email text provided.")
    else:
        st.divider()
        preview = body[:200] + ("..." if len(body) > 200 else "")
        st.markdown(f"**Subject:** {subject if subject else '_(none provided)_'}")

        vectoriser, rf_model, directions, tokeniser, db_model, device = get_models()

        with st.spinner("Running both models..."):
            rf_pred, rf_conf, rf_latency = predict_rf(vectoriser, rf_model, text)
            rf_top = explain_rf(vectoriser, rf_model, directions, text)
            db_pred, db_conf, db_latency, db_top = predict_and_explain_db(
                tokeniser, db_model, device, text

        col_rf, col_db = st.columns(2)

        with col_rf:
            st.subheader("Random Forest")

            m2.metric("Confidence", f"{rf_conf:.4f}")
            m3.metric("Latency", f"{rf_latency:.2f} ms")
            if rf_top:
                st.markdown("**Top contributing features:**")
                st.table([
                    {"Feature": f, "Importance": f"{imp:.4f}", "Indicates": d}
                    for f, imp, d in rf_top
                ])
            else:
                st.info("No recognised training vocabulary found in this email.")

        with col_db:
            st.subheader("DistilBERT")
            m1, m2, m3 = st.columns(3)
            m1.metric("Prediction", db_pred.upper())
            m2.metric("Confidence", f"{db_conf:.4f}")
            m3.metric("Latency", f"{db_latency:.2f} ms")
            if db_top:
                st.markdown("**Top salient tokens (gradient x input):**")
                st.table([
                    {"Token": t, "Saliency": f"{s:.4f}"} for t, s in db_top
                ])
            else:
                st.info("No salient tokens found.")

        st.divider()
        if rf_pred == db_pred:
            st.success(f"Both models agree: **{rf_pred.upper()}**")
        else:
            st.warning(
                f"Models disagree — Random Forest says **{rf_pred}**, "
                f"DistilBERT says **{db_pred}**"
            )

        speed_ratio = db_latency / rf_latency if rf_latency > 0 else 0
        st.caption(f"Random Forest is {speed_ratio:.1f}x faster than DistilBERT on this email.")
